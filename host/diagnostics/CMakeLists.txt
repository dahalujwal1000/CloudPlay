set(CLOUDPLAY_NVENC_INCLUDE_DIR "" CACHE PATH "Directory containing NVIDIA nvEncodeAPI.h")

if(CLOUDPLAY_NVENC_INCLUDE_DIR)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The NVENC readiness diagnostic currently supports Linux only")
  endif()
  if(NOT EXISTS "${CLOUDPLAY_NVENC_INCLUDE_DIR}/nvEncodeAPI.h")
    message(FATAL_ERROR "CLOUDPLAY_NVENC_INCLUDE_DIR must contain nvEncodeAPI.h")
  endif()
  add_executable(cloudplay_nvenc_probe nvenc_probe.cpp)
  target_include_directories(cloudplay_nvenc_probe SYSTEM PRIVATE "${CLOUDPLAY_NVENC_INCLUDE_DIR}")
  target_link_libraries(cloudplay_nvenc_probe PRIVATE cloudplay_options ${CMAKE_DL_LIBS})
  if(BUILD_TESTING)
    add_test(NAME nvenc.probe_help COMMAND cloudplay_nvenc_probe --help)
    add_test(NAME nvenc.probe_invalid_cli COMMAND cloudplay_nvenc_probe --invalid)
    set_tests_properties(nvenc.probe_invalid_cli PROPERTIES WILL_FAIL TRUE)
    set(failure_reasons driver_api_too_old version_query_failed interface_creation_failed required_function_unavailable driver_entry_point_unavailable)
    set(mock_case 0)
    foreach(reason IN LISTS failure_reasons)
      math(EXPR mock_case "${mock_case} + 1")
      add_library(nvenc_mock_${mock_case} SHARED tests/mock_driver.cpp)
      target_include_directories(nvenc_mock_${mock_case} SYSTEM PRIVATE "${CLOUDPLAY_NVENC_INCLUDE_DIR}")
      target_link_libraries(nvenc_mock_${mock_case} PRIVATE cloudplay_options)
      target_compile_definitions(nvenc_mock_${mock_case} PRIVATE CLOUDPLAY_MOCK_CASE=${mock_case})
      set_target_properties(nvenc_mock_${mock_case} PROPERTIES OUTPUT_NAME nvidia-encode SUFFIX ".so.1" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/mock-${mock_case}")
      add_test(NAME nvenc.${reason} COMMAND "${CMAKE_COMMAND}"
        "-DPROBE=$<TARGET_FILE:cloudplay_nvenc_probe>"
        "-DMOCK_DIR=$<TARGET_FILE_DIR:nvenc_mock_${mock_case}>"
        "-DEXPECT_REASON=${reason}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_failure.cmake")
    endforeach()
  endif()
endif()

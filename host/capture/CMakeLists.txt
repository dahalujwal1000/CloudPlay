if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  option(CLOUDPLAY_LINUX_CAPTURE "Build the Wayland portal/PipeWire GPU capture backend" OFF)
  if(CLOUDPLAY_LINUX_CAPTURE)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3>=0.3.65)
    pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
    pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
    pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)
    add_library(cloudplay_linux_capture linux/pipewire_capture.cpp linux/portal_session.cpp linux/gpu_import.cpp)
    add_library(CloudPlay::LinuxCapture ALIAS cloudplay_linux_capture)
    target_link_libraries(cloudplay_linux_capture PUBLIC cloudplay_capture_policy PRIVATE cloudplay_options PkgConfig::PIPEWIRE PkgConfig::GIO PkgConfig::EGL PkgConfig::DRM)
    add_executable(cloudplay_linux_capture_probe linux/capture_probe.cpp)
    target_link_libraries(cloudplay_linux_capture_probe PRIVATE CloudPlay::LinuxCapture cloudplay_options)
    if(BUILD_TESTING)
      add_executable(cloudplay_linux_capture_tests tests/linux_capture_tests.cpp)
      target_link_libraries(cloudplay_linux_capture_tests PRIVATE CloudPlay::LinuxCapture cloudplay_options)
      add_test(NAME capture.linux_lifecycle COMMAND "${CMAKE_COMMAND}" -E env --unset=DBUS_SESSION_BUS_ADDRESS --unset=XDG_SESSION_TYPE $<TARGET_FILE:cloudplay_linux_capture_tests>)
      add_test(NAME capture.linux_invalid_cli COMMAND cloudplay_linux_capture_probe --seconds 0)
      set_tests_properties(capture.linux_invalid_cli PROPERTIES WILL_FAIL TRUE)
      add_executable(cloudplay_portal_lifecycle_tests tests/portal_lifecycle_tests.cpp)
      target_link_libraries(cloudplay_portal_lifecycle_tests PRIVATE CloudPlay::LinuxCapture PkgConfig::GIO cloudplay_options)
      add_test(NAME capture.portal_shutdown COMMAND cloudplay_portal_lifecycle_tests)
      set_tests_properties(capture.portal_shutdown PROPERTIES TIMEOUT 20)
    endif()
  endif()
endif()

  endif()
  add_library(cloudplay_capture windows/window_capture.cpp)
  add_library(CloudPlay::Capture ALIAS cloudplay_capture)
  target_link_libraries(cloudplay_capture PUBLIC cloudplay_capture_policy PRIVATE cloudplay_options d3d11 dxgi windowsapp user32)
  target_compile_definitions(cloudplay_capture PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  add_executable(cloudplay_capture_probe windows/capture_probe.cpp)
  target_link_libraries(cloudplay_capture_probe PRIVATE CloudPlay::Capture cloudplay_options)
  if(BUILD_TESTING)
    add_executable(cloudplay_capture_validation_tests windows/capture_validation_tests.cpp)

  endif()

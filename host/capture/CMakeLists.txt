if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  option(CLOUDPLAY_LINUX_CAPTURE "Build the Wayland portal/PipeWire GPU capture backend" OFF)
  if(CLOUDPLAY_LINUX_CAPTURE)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3>=0.3.65)
    pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
    pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
    pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)
    include(CheckCXXSourceCompiles)
    set(CMAKE_REQUIRED_INCLUDES ${PIPEWIRE_INCLUDE_DIRS})
    check_cxx_source_compiles("#include <spa/buffer/meta.h>\nint main() { return sizeof(spa_meta_sync_timeline) + SPA_META_SyncTimeline; }" CLOUDPLAY_HAVE_SPA_SYNC_TIMELINE)
    unset(CMAKE_REQUIRED_INCLUDES)
    add_library(cloudplay_linux_capture linux/pipewire_capture.cpp linux/portal_session.cpp linux/gpu_import.cpp)
    add_library(CloudPlay::LinuxCapture ALIAS cloudplay_linux_capture)
    target_link_libraries(cloudplay_linux_capture PUBLIC cloudplay_capture_policy PRIVATE cloudplay_options PkgConfig::PIPEWIRE PkgConfig::GIO PkgConfig::EGL PkgConfig::DRM)
    if(CLOUDPLAY_HAVE_SPA_SYNC_TIMELINE)
      target_compile_definitions(cloudplay_linux_capture PRIVATE CLOUDPLAY_HAVE_SPA_SYNC_TIMELINE=1)
    endif()
    add_executable(cloudplay_linux_capture_probe linux/capture_probe.cpp)
    target_link_libraries(cloudplay_linux_capture_probe PRIVATE CloudPlay::LinuxCapture cloudplay_options)
    find_package(PNG QUIET)
    add_library(cloudplay_capture_snapshot_diagnostic linux/cpu_snapshot.cpp)
    target_link_libraries(cloudplay_capture_snapshot_diagnostic PUBLIC cloudplay_capture_policy PRIVATE cloudplay_options PkgConfig::DRM PkgConfig::PIPEWIRE)
    target_link_libraries(cloudplay_linux_capture_probe PRIVATE cloudplay_capture_snapshot_diagnostic)
    target_sources(cloudplay_linux_capture_probe PRIVATE linux/capture_diagnostics.cpp)
    if(PNG_FOUND)
      target_link_libraries(cloudplay_capture_snapshot_diagnostic PRIVATE PNG::PNG)
      target_compile_definitions(cloudplay_capture_snapshot_diagnostic PRIVATE CLOUDPLAY_HAVE_PNG=1)
      target_compile_definitions(cloudplay_linux_capture_probe PRIVATE CLOUDPLAY_HAVE_PNG=1)
    endif()
    if(BUILD_TESTING)
      add_executable(cloudplay_cpu_snapshot_tests tests/cpu_snapshot_tests.cpp)
      target_link_libraries(cloudplay_cpu_snapshot_tests PRIVATE cloudplay_capture_snapshot_diagnostic cloudplay_options PkgConfig::DRM)
      if(PNG_FOUND)
        target_compile_definitions(cloudplay_cpu_snapshot_tests PRIVATE CLOUDPLAY_HAVE_PNG=1)
      endif()
      add_test(NAME capture.cpu_snapshot COMMAND cloudplay_cpu_snapshot_tests)
      add_executable(cloudplay_linux_capture_tests tests/linux_capture_tests.cpp)
      target_link_libraries(cloudplay_linux_capture_tests PRIVATE CloudPlay::LinuxCapture cloudplay_options)
      add_test(NAME capture.linux_lifecycle COMMAND "${CMAKE_COMMAND}" -E env --unset=DBUS_SESSION_BUS_ADDRESS --unset=XDG_SESSION_TYPE $<TARGET_FILE:cloudplay_linux_capture_tests>)
      add_test(NAME capture.linux_invalid_cli COMMAND cloudplay_linux_capture_probe --seconds 0)
      set_tests_properties(capture.linux_invalid_cli PROPERTIES WILL_FAIL TRUE)
      add_test(NAME capture.diagnostics_invalid_frames COMMAND cloudplay_linux_capture_probe --capture-diagnostics --frames 0)
      add_test(NAME capture.diagnostics_missing_value COMMAND cloudplay_linux_capture_probe --capture-diagnostics --reference)
      set_tests_properties(capture.diagnostics_invalid_frames capture.diagnostics_missing_value PROPERTIES WILL_FAIL TRUE)
      add_executable(cloudplay_portal_lifecycle_tests tests/portal_lifecycle_tests.cpp)
      target_link_libraries(cloudplay_portal_lifecycle_tests PRIVATE CloudPlay::LinuxCapture PkgConfig::GIO cloudplay_options)
      add_test(NAME capture.portal_shutdown COMMAND cloudplay_portal_lifecycle_tests)
      set_tests_properties(capture.portal_shutdown PROPERTIES TIMEOUT 20)
    endif()
  endif()
endif()
